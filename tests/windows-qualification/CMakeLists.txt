cmake_minimum_required(VERSION 3.10)
project(ShalimarQualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 98)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(SHALIMAR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${SHALIMAR_ROOT}" compiler EXCLUDE_FROM_ALL)
get_target_property(COMPILER_SOURCES shalimarc SOURCES)
set(QUALIFICATION_SOURCES)
foreach(source IN LISTS COMPILER_SOURCES)
    if(NOT source STREQUAL "src/main.cpp")
        list(APPEND QUALIFICATION_SOURCES "${SHALIMAR_ROOT}/${source}")
    endif()
endforeach()
add_library(qualification_core STATIC ${QUALIFICATION_SOURCES})
target_include_directories(qualification_core PUBLIC "${SHALIMAR_ROOT}/include")
enable_testing()
file(GLOB TEST_SOURCES "${SHALIMAR_ROOT}/tests/*.cpp")
foreach(source IN LISTS TEST_SOURCES)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_link_libraries(${name} PRIVATE qualification_core)
    # These supplied tests use assert: retain assertions in Release too.
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    if(name STREQUAL "rev35_coff_relocations")
        add_test(NAME ${name} COMMAND ${name} "${CMAKE_CURRENT_BINARY_DIR}/relocations-$<CONFIG>.obj")
    else()
        add_test(NAME ${name} COMMAND ${name})
    endif()
endforeach()
